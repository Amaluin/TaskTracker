ТЕСТОВЫЕ ДАННЫЕ
---------------
INSERT INTO tasks (user_id, title, description, status, priority, due_at)
VALUES ('test-user-1', 'Тест', 'Описание', 'pending', 'high', NOW() + INTERVAL '1 day');

SELECT * FROM tasks WHERE user_id = 'test-user-1';

DELETE FROM tasks WHERE user_id = 'test-user-1';

ПОЛЕЗНЫЕ ЗАПРОСЫ
----------------
-- Все задачи
SELECT * FROM tasks ORDER BY created_at DESC;

-- Статистика
SELECT 
  COUNT(*) as total,
  COUNT(*) FILTER (WHERE status = 'pending') as pending,
  COUNT(*) FILTER (WHERE status = 'accepted') as accepted
FROM tasks;

-- Просроченные
SELECT * FROM overdue_tasks;

-- Ближайшие дедлайны
SELECT * FROM upcoming_tasks;

DOCKER КОМАНДЫ
--------------
# Статус
docker-compose -f docker-compose.supabase.yml ps

# Логи
docker-compose -f docker-compose.supabase.yml logs -f

# Перезапуск
docker-compose -f docker-compose.supabase.yml restart

# Остановка
docker-compose -f docker-compose.supabase.yml down

# Запуск
docker-compose -f docker-compose.supabase.yml up -d

БЭКАП БД
--------
# Создать
docker exec supabase-db pg_dump -U postgres postgres > backup.sql

# Восстановить
docker exec -i supabase-db psql -U postgres postgres < backup.sql

TROUBLESHOOTING
---------------
