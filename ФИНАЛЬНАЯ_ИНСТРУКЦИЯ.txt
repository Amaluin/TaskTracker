ШАГ 5: ПРОВЕРИТЬ СТРУКТУРУ БД
═══════════════════════════════════════════════════════════

Выполните в SQL Editor:

-- Проверка 1: Таблица user_profiles существует
SELECT table_name 
FROM information_schema.tables 
WHERE table_schema = 'public' 
  AND table_name = 'user_profiles';
-- Должна вернуть: user_profiles

-- Проверка 2: Тип user_id изменен на UUID
SELECT column_name, data_type 
FROM information_schema.columns 
WHERE table_name = 'tasks' 
  AND column_name = 'user_id';
-- Должна вернуть: uuid

-- Проверка 3: Внешний ключ на auth.users
SELECT
  ccu.table_schema || '.' || ccu.table_name AS references
FROM information_schema.table_constraints AS tc
JOIN information_schema.constraint_column_usage AS ccu
  ON ccu.constraint_name = tc.constraint_name
WHERE tc.table_name = 'tasks'
  AND tc.constraint_name = 'tasks_user_id_fkey';
-- Должна вернуть: auth.users
